# 双餐谱实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将现有的单餐谱「今天吃什么」拆分为「日常」和「改善伙食」两个独立餐谱，各自拥有独立的食物清单和抽奖功能。

**Architecture:** 在单个 HTML 文件中，通过 CSS 变量和 data 属性区分两个餐谱面板。JS 层将 foods 数组重构为 `{ daily: [], premium: [] }` 对象，所有渲染/抽奖函数接受 menuKey 参数。localStorage 使用两个独立 key。

**Tech Stack:** 纯 HTML/CSS/JS，无外部依赖，单文件应用。

```javascript
    // ===== Data Layer =====
    const STORAGE_KEY_DAILY = 'food-list-daily';
    const STORAGE_KEY_PREMIUM = 'food-list-premium';
    const LEGACY_KEY = 'food-list';

    function loadFoods(key, defaults) {
      try {
        const raw = localStorage.getItem(key);
        if (raw) {
          const parsed = JSON.parse(raw);
          if (Array.isArray(parsed)) return parsed.filter(f => typeof f === 'string' && f.trim());
        }
      } catch (e) { /* localStorage corrupt or unavailable */ }
      return [...defaults];
    }

    function saveFoods(key, foods) {
      try {
        localStorage.setItem(key, JSON.stringify(foods));
      } catch (e) {
        showToast('存储空间不足，请清理一些数据');
      }
    }

    // Migrate legacy data on first load
    (function migrateLegacy() {
      try {
        const legacy = localStorage.getItem(LEGACY_KEY);
        if (legacy) {
          const parsed = JSON.parse(legacy);
          if (Array.isArray(parsed) && parsed.length > 0) {
            // Only migrate if daily key doesn't exist yet
            if (!localStorage.getItem(STORAGE_KEY_DAILY)) {
              localStorage.setItem(STORAGE_KEY_DAILY, JSON.stringify(parsed));
            }
          }
          localStorage.removeItem(LEGACY_KEY);
        }
      } catch (e) { /* ignore */ }
    })();

    // In-memory state
    const DEFAULT_DAILY = ['兰州拉面', '黄焖鸡米饭', '沙县小吃', '麦当劳', '麻辣烫'];
    const DEFAULT_PREMIUM = [];

    let foods = {
      daily: loadFoods(STORAGE_KEY_DAILY, DEFAULT_DAILY),
      premium: loadFoods(STORAGE_KEY_PREMIUM, DEFAULT_PREMIUM),
    };

    function persistMenu(menuKey) {
      const key = menuKey === 'daily' ? STORAGE_KEY_DAILY : STORAGE_KEY_PREMIUM;
      saveFoods(key, foods[menuKey]);
    }
```

```javascript
    function renderList(menuKey) {
      const refs = domRefs[menuKey];
```

```javascript
    // ===== Modal: Add / Edit =====
    function openAddModal(menuKey) {
      currentMenu = menuKey;
      editIndex = -1;
      const label = menuKey === 'daily' ? '日常' : '改善伙食';
      modalTitle.textContent = `添加食物 — ${label}`;
      modalInput.value = '';
      modalError.textContent = '';
      modalOverlay.classList.add('active');
      setTimeout(() => modalInput.focus(), 150);
    }

    function openEditModal(menuKey, index) {
      currentMenu = menuKey;
      editIndex = index;
      const label = menuKey === 'daily' ? '日常' : '改善伙食';
      modalTitle.textContent = `编辑食物 — ${label}`;
      modalInput.value = foods[menuKey][index];
      modalError.textContent = '';
      modalOverlay.classList.add('active');
      setTimeout(() => { modalInput.focus(); modalInput.select(); }, 150);
    }

    function closeModal() {
      modalOverlay.classList.remove('active');
      modalInput.value = '';
      editIndex = -1;
    }

    function confirmModal() {
      const name = modalInput.value.trim();
      if (!name) {
        modalError.textContent = '名称不能为空';
        modalInput.focus();
        return;
      }

      if (editIndex === -1) {
        foods[currentMenu].push(name);
        showToast(`已添加「${name}」`);
      } else {
        const oldName = foods[currentMenu][editIndex];
        foods[currentMenu][editIndex] = name;
        showToast(`已更新「${oldName}」→「${name}」`);
      }

      persistMenu(currentMenu);
      renderList(currentMenu);
      closeModal();
    }
```

```javascript
    // ===== Lottery =====
    function resetLottery(menuKey) {
      const refs = domRefs[menuKey];
      refs.lotteryResult.textContent = '';
      refs.lotteryResult.classList.remove('bounce');
      refs.lotteryBox.classList.remove('result');
      refs.btnLottery.textContent = '🎲 开抽！';
      if (foods[menuKey].length > 0) refs.btnLottery.disabled = false;
    }

    function isRolling(menuKey) {
      return menuKey === 'daily' ? isRollingDaily : isRollingPremium;
    }

    function setRolling(menuKey, val) {
      if (menuKey === 'daily') isRollingDaily = val;
      else isRollingPremium = val;
    }

    function startLottery(menuKey) {
      const refs = domRefs[menuKey];
      const list = foods[menuKey];
      const count = list.length;
      if (count === 0 || isRolling(menuKey)) return;

      setRolling(menuKey, true);
      refs.btnLottery.disabled = true;
      refs.lotteryResult.classList.remove('bounce');
      refs.lotteryBox.classList.add('rolling');
      refs.lotteryBox.classList.remove('result');

      const winnerIndex = Math.floor(Math.random() * count);
      const winner = list[winnerIndex];

      const fastSteps = 18;
      const slowSteps = 6;
      const sequence = [];

      for (let i = 0; i < fastSteps; i++) {
        sequence.push({ name: list[Math.floor(Math.random() * count)], delay: 80 });
      }

      for (let i = 0; i < slowSteps; i++) {
        const progress = i / slowSteps;
        const delay = 80 + (300 - 80) * progress;
        const name = i === slowSteps - 1 ? winner : list[Math.floor(Math.random() * count)];
        sequence.push({ name, delay: Math.round(delay) });
      }

      let step = 0;
      function nextStep() {
        if (step >= sequence.length) {
          finishLottery(menuKey, winner);
          return;
        }
        refs.lotteryResult.textContent = sequence[step].name;
        step++;
        setTimeout(nextStep, sequence[step - 1].delay);
      }
      nextStep();
    }

    function finishLottery(menuKey, winner) {
      const refs = domRefs[menuKey];
      refs.lotteryBox.classList.remove('rolling');
      refs.lotteryBox.classList.add('result');
      refs.lotteryResult.textContent = winner;
      refs.lotteryResult.classList.add('bounce');
      refs.btnLottery.textContent = '🔄 再来一次';
      refs.btnLottery.disabled = false;
      setRolling(menuKey, false);
      spawnConfetti(refs.lotteryBox);
    }

    // ===== Confetti =====
    function spawnConfetti(box) {
      const emojis = ['🎉', '✨', '🍜', '🍛', '🥟', '🍔', '🍕', '🥡'];
      const boxRect = box.getBoundingClientRect();

      for (let i = 0; i < 8; i++) {
        const el = document.createElement('span');
        el.className = 'confetti';
        el.textContent = emojis[Math.floor(Math.random() * emojis.length)];
        el.style.left = (boxRect.left + Math.random() * boxRect.width) + 'px';
        el.style.top  = (boxRect.top + boxRect.height / 2 + Math.random() * 30) + 'px';
        el.style.animationDelay = (Math.random() * 0.3) + 's';
        el.style.fontSize = (20 + Math.random() * 16) + 'px';
        document.body.appendChild(el);
        setTimeout(() => el.remove(), 1500);
      }
    }

    btnLotteryDaily.addEventListener('click', () => handleLotteryClick('daily'));
    btnLotteryPremium.addEventListener('click', () => handleLotteryClick('premium'));

    function handleLotteryClick(menuKey) {
      const refs = domRefs[menuKey];
      const list = foods[menuKey];
      if (list.length === 0) return;
      if (list.length === 1) {
        refs.lotteryBox.classList.add('result');
        refs.lotteryResult.textContent = list[0];
        refs.lotteryResult.classList.add('bounce');
        refs.btnLottery.textContent = '🔄 就这一个，别犹豫了！';
        return;
      }
      startLottery(menuKey);
    }
```
